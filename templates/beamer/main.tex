\section{Überblick}

\begin{frame}{Warum ein eigener Editor?}
  \begin{block}{Vorteile}
    \begin{itemize}
      \item Volle Kontrolle über Werkzeuge und Pakete
      \item Lokale Daten – keine Cloud-Abhängigkeit
      \item SyncTeX: Klick ins PDF springt in den Quelltext
    \end{itemize}
  \end{block}

  \begin{alertblock}{Nachteil}
    Kein Multi-User-Kollaborationsmodus (by design).
  \end{alertblock}
\end{frame}
